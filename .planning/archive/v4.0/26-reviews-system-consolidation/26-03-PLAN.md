Use `IF EXISTS` on both drops to be idempotent.

```typescript
// Migration template (adapt to match project's migration style):
export async function up(db: any): Promise<void> {
  await db.query('DROP TABLE IF EXISTS review_helpfulness CASCADE');
  await db.query('DROP TABLE IF EXISTS reviews CASCADE');
}

export async function down(db: any): Promise<void> {
  // Recreate tables for rollback
  await db.query(`
    CREATE TABLE IF NOT EXISTS reviews (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      user_id UUID NOT NULL REFERENCES users(id),
      venue_id UUID REFERENCES venues(id),
      band_id UUID REFERENCES bands(id),
      rating INTEGER NOT NULL CHECK (rating >= 1 AND rating <= 5),
      content TEXT,
      helpful_count INTEGER DEFAULT 0,
      owner_response TEXT,
      owner_response_at TIMESTAMPTZ,
      is_hidden BOOLEAN DEFAULT false,
      created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
      updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
    )
  `);
  await db.query(`
    CREATE TABLE IF NOT EXISTS review_helpfulness (
      id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
      review_id UUID NOT NULL REFERENCES reviews(id) ON DELETE CASCADE,
      user_id UUID NOT NULL REFERENCES users(id),
      is_helpful BOOLEAN NOT NULL,
      created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
      UNIQUE(review_id, user_id)
    )
  `);
}
```
